等价 SQL

WITH RECURSIVE reach AS (
  SELECT dst, 1 AS d, CONCAT(src,'->',dst) AS path
  FROM edges WHERE src = 42
  UNION ALL
  SELECT e.dst, r.d+1, CONCAT(r.path,'->',e.dst)
  FROM edges e JOIN reach r ON e.src = r.dst
  WHERE r.d < 3
) SELECT * FROM reach;

import daft
df = daft.read_lance("s3://lake/papers")
df = (df.where(df["doc_type"] == "paper")
        .select(df["title"], df["quality_score"])
        .sort(df["quality_score"], desc=True)
        .collect())  # 懒求值，collect() 时才执行
链式算子：sort → filter → groupby → join → sql → pivot → explode → sample。支持多模态 decode（URL→图像→AI 函数）与 Ray 分布式。

-- 跨存储：Lance 表 JOIN MinIO 的 parquet，物化成视图
CREATE MATERIALIZED VIEW mv_monthly_quality AS
SELECT doc_type, DATE_TRUNC('month', created_at) AS month,
       AVG(quality_score) AS avg_q
FROM papers
GROUP BY doc_type, month
WITH TTL = '7d', ART = 'incremental';
