### 1.1 Lake Context Manager + Shutdown 修复

**文件**：`arrow_lake/__init__.py`

| 改动 | 说明 |
|------|------|
| 增加 `__enter__` / `__exit__` | 支持 `with Lake() as lake:` 自动清理 |
| 增加 `__del__` | 发出 `ResourceWarning` 提醒未 shutdown |
| 修复 `shutdown()` 异步清理 | 收集 async tasks → `asyncio.gather` 等待 |
| `_get_component` 加锁 | `threading.Lock` 保护 factory 调用 |

```python
class Lake(...):
    _COMPONENT_LOCK = threading.Lock()

    def _get_component(self, key, factory):
        if key not in self._components:
            with self._COMPONENT_LOCK:
                if key not in self._components:  # double-check
                    self._components[key] = factory()
        return self._components[key]

    def __enter__(self):  return self
    def __exit__(self, *exc):  self.shutdown()

    def __del__(self):
        if self._components:
            import warnings
            warnings.warn("Lake not shut down — call lake.shutdown() or use 'with'", ResourceWarning, stacklevel=1)
```

### 2.3 多轮对话历史注入

**文件**：`arrow_lake/rag/pipeline.py`

```python
def _build_messages(self, question, context_text, template_name=None, history=None):
    messages = []
    # System prompt
    if self._config.system_prompt:
        messages.append(LLMMessage(role="system", content=self._config.system_prompt))

    # History (recent N turns, respecting token budget)
    if history:
        history_budget = int(self._context_window_tokens * 0.2)  # 20% for history
        for turn in history[-6:]:  # 最近 3 轮 (每轮 user+assistant)
            if count_tokens(turn["question"]) + count_tokens(turn["answer"]) < history_budget:
                messages.append(LLMMessage(role="user", content=turn["question"]))
                messages.append(LLMMessage(role="assistant", content=turn["answer"]))

    # Current query with context
    prompt = template.render(context=context_text, question=_sanitize(question))
    messages.append(LLMMessage(role="user", content=prompt))
    return messages
```

### 2.4 Context Window 排序优化

**文件**：`arrow_lake/rag/context.py`

修改 `add_chunk` 流程：先收集所有 candidate chunks，按 score 降序排序后再填充 token budget。避免低分 chunk 先占满预算。

```python
class ContextWindow:
    def finalize(self) -> None:
        """Sort chunks by score descending, then apply budget."""
        self._chunks.sort(key=lambda c: c.score, reverse=True)
        # Re-apply token budget in score order
        trimmed = []
        tokens = 0
        for chunk in self._chunks:
            chunk_tokens = count_tokens(chunk.text)
            if tokens + chunk_tokens <= self._token_budget:
                trimmed.append(chunk)
                tokens += chunk_tokens
        self._chunks = trimmed
        self._current_tokens = tokens
```

def _truncate_at_sentence(text: str, max_chars: int) -> str:
    truncated = text[:max_chars]
    # 找到最后一个句子结束位置
    for sep in ('。', '！', '？', '.', '!', '?', '\n'):
        idx = truncated.rfind(sep)
        if idx > len(truncated) * 0.5:  # 至少保留 50%

### 4.3 Model Fallback Chain

**文件**：`arrow_lake/rag/provider.py`

```python
class FallbackProvider(BaseLLMProvider):
    """Provider with cascading fallback on failure."""

    def __init__(self, providers: list[BaseLLMProvider]):
        self._providers = providers

    async def generate(self, messages):
        last_error = None
        for provider in self._providers:
            try:
                return await provider.generate(messages)
            except RAGError as e:
                last_error = e
                continue
        raise last_error
```

### 4.8 GraphRAG 实体缓存

**文件**：`arrow_lake/rag/graph_rag.py`

```python
from functools import lru_cache
import hashlib

class QuestionEntityCache:
    def __init__(self, ttl: int = 300, max_size: int = 1000):
        self._cache: dict[str, tuple[float, list[str]]] = {}
        self._ttl = ttl
        self._max_size = max_size

    def get(self, question: str) -> list[str] | None:
        key = hashlib.md5(question.encode()).hexdigest()
        entry = self._cache.get(key)
        if entry and time.time() - entry[0] < self._ttl:
            return entry[1]
        return None

    def set(self, question: str, entities: list[str]):
        key = hashlib.md5(question.encode()).hexdigest()
        if len(self._cache) >= self._max_size:
            oldest = min(self._cache, key=lambda k: self._cache[k][0])
            del self._cache[oldest]
        self._cache[key] = (time.time(), entities)
```
